cmake_minimum_required(VERSION 3.22)
project(jackal_description)

set(PATHCOVER_EXAMPLE "quadrotor" CACHE STRING "PathCover example to build (quadrotor or jackal)")
if(NOT PATHCOVER_EXAMPLE STREQUAL "quadrotor" AND NOT PATHCOVER_EXAMPLE STREQUAL "jackal")
  message(FATAL_ERROR "PATHCOVER_EXAMPLE must be 'quadrotor' or 'jackal' (got '${PATHCOVER_EXAMPLE}')")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release)
add_compile_options(-O3 -w)

if(PATHCOVER_EXAMPLE STREQUAL "jackal")
  find_package(catkin REQUIRED COMPONENTS geometry_msgs nav_msgs roscpp sensor_msgs tf)
  catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS geometry_msgs nav_msgs roscpp sensor_msgs tf)
  include_directories(include ${catkin_INCLUDE_DIRS})
  file(GLOB_RECURSE SRC_FILES "src/*.cpp")
  add_executable(tf_publisher ${SRC_FILES})
  target_link_libraries(tf_publisher ${catkin_LIBRARIES})
  install(TARGETS tf_publisher RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
else()
  find_package(catkin REQUIRED)
  catkin_package()
endif()


install(DIRECTORY urdf DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY meshes DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
